# ============================================================================
#  CMP-Agent-Project —— RAG 微服务依赖（**独立环境**，Python 3.11）
#
#  为什么单独一份：
#    1. chromadb 硬依赖 onnxruntime，而 onnxruntime 在 Python 3.15 上没有 wheel；
#    2. BGE-M3（2.27 GB）与本向量库是用户既有资产，只引用绝对路径、不随仓库分发；
#    3. 微服务由 启动RAG服务.bat 以 `py -3.11` 启动，与主环境完全隔离。
#
#  安装：py -3.11 -m pip install -r requirements-rag.txt
#  已在本机验证：
#    chromadb 1.5.9 | langchain-huggingface 1.2.3 | langchain-text-splitters 1.1.3
#    langchain-core 1.7.4 | sentence-transformers 6.2.0 | transformers 5.22.1
#    tokenizers 0.23.3 | torch 2.15.0
# ============================================================================

chromadb==1.5.9
langchain-core==1.7.4
langchain-huggingface==1.2.3
langchain-text-splitters==1.1.3
sentence-transformers==6.2.0
transformers==5.22.1
tokenizers==0.23.3

# 与主环境共用同一 CPU 版 torch（务必版本一致，否则 sentence-transformers 会重装 GPU 版）
torch==2.15.0
numpy==2.4.6

# 说明：onnxruntime 只在 chromadb 自带默认 embedding（all-MiniLM-L6-v2）时才会用到；
# 本项目所有向量都由本地 BGE-M3 产生，故**不需要**安装它。
